# CLI shell completion scripts

Generated from click's built-in completion support.
Install per-shell:

  bash:   source completion/downloadkit.bash
  zsh:    ln -s completion/downloadkit.zsh ~/.zsh/completion/_downloadkit
  fish:   downloadkit completion fish > ~/.config/fish/completions/downloadkit.fish
  pwsh:   downloadkit completion powershell | Out-File $PROFILE
